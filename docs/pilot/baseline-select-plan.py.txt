"""PROPOSAL ONLY: pure metadata-to-SELECT generator; no DB/network/capture/restore calls.

Reviewed for the pre-install sandbox's twelve native tables. A different phase needs review.
Feed approved private catalog JSON; send emitted SQL to one read-only client connection later.
"""
import json
import re
import sys

PREFIX = 'TXGEh_'
NATIVE = {'commentmeta', 'comments', 'links', 'options', 'postmeta', 'posts',
          'term_relationships', 'term_taxonomy', 'termmeta', 'terms', 'usermeta', 'users'}


def identifier(value):
    if not isinstance(value, str) or not re.fullmatch(r'[A-Za-z0-9_]+', value):
        raise ValueError('Unreviewed identifier')
    return '`' + value + '`'


def cell(column):
    name = identifier(column)
    raw = 'CAST(' + name + ' AS BINARY)'
    return "IF(" + name + " IS NULL,'N;',CONCAT('V',OCTET_LENGTH(" + raw + "),':',LOWER(HEX(" + raw + ")),';'))"


def generate(catalog):
    tables = {}
    for column in catalog['columns']:
        name = column['table']
        identifier(name)
        identifier(column['column'])
        if not name.startswith(PREFIX) or column['engine'] != 'InnoDB':
            raise ValueError('Wrong source/storage')
        tables.setdefault(name, []).append(column)
    if set(tables) != {PREFIX + name for name in NATIVE}:
        raise ValueError('Different source phase/table set; review before capture')
    lines = ["SET SESSION time_zone='+00:00';",
             'SET SESSION TRANSACTION ISOLATION LEVEL REPEATABLE READ;',
             'SET SESSION TRANSACTION READ ONLY;',
             'START TRANSACTION WITH CONSISTENT SNAPSHOT;',
             "SELECT JSON_OBJECT('kind','capture','version',1,'database',DATABASE(),'prefix','TXGEh_','captured_at',UTC_TIMESTAMP(6));"]
    for name, columns in sorted(tables.items()):
        columns = sorted(columns, key=lambda c: c['ordinal'])
        if [c['ordinal'] for c in columns] != list(range(1, len(columns)+1)) or len({c['column'] for c in columns}) != len(columns):
            raise ValueError('Duplicate/missing columns')
        primary = sorted((c for c in columns if c['pk_order'] is not None), key=lambda c: c['pk_order'])
        if not primary or [c['pk_order'] for c in primary] != list(range(1, len(primary)+1)):
            raise ValueError('Missing/ambiguous ordered primary key')
        logical = name[len(PREFIX):]
        # Same length-framed bytes as the synthetic local reference; no JSON-spacing dependency.
        domain = 'V' + str(len(logical.encode())) + ':' + logical.encode().hex() + ';'
        key = "SHA2(CONCAT('CSFBASELINE1',CHAR(0),'" + domain + "'," + ','.join(cell(c['column']) for c in primary) + '),256)'
        value = "SHA2(CONCAT('CSFBASELINE1',CHAR(0)," + ','.join(cell(c['column']) for c in columns) + '),256)'
        if logical == 'options' and 'option_name' not in {c['column'] for c in columns}:
            raise ValueError('Options identity label missing')
        label = '`option_name`' if logical == 'options' else 'NULL'
        lines.append("SELECT JSON_OBJECT('table','" + logical + "','key_hash'," + key + ",'value_hash'," + value + ",'label'," + label + ') FROM ' + identifier(name) + ' ORDER BY ' + ','.join(identifier(c['column']) for c in primary) + ';')
    lines.append('COMMIT;')
    return '\n'.join(lines) + '\n'


if __name__ == '__main__':
    if len(sys.argv) != 2:
        raise SystemExit('Private catalog JSON path required; this program never connects to a database')
    with open(sys.argv[1], encoding='utf-8') as stream:
        print(generate(json.load(stream)), end='')
